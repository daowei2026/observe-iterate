224: loadPackedProductRows :: PyAny -> [Text] -> FaoModel (FrpEvent [(Text, NumericRow)])
225: loadPackedProductRows source prefixes = do
226:   (fullDir, groups) <- loadPackedProductMetadata source prefixes
227:   target <- getTarName source
228:   events <-
229:     forM (Map.toList groups) $ \(resultKey, fields) -> do
230:       let ?hisBack = BackNumber 0
231:       event <-
232:         lift $
233:           newAdtFrpEvent4His @PackedNumericRowsHistoryData @NumericRow
234:             (packEventName source target ("__eaw_occurrence_row__." <> resultKey))
235:             (BackNumber 0)
236:             (loadPackedNumericRows fullDir resultKey fields)
237:       pure ((\tick -> [(resultKey, atTickValue tick)]) <$> event)
